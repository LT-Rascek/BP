scripted_effect remove_character_flag_if_has_it_effect = {
	if = {
		limit = { has_character_flag = $FLAG$ }
		remove_character_flag = $FLAG$
	}

#Culture diverges and adopts traditions
migration_situation_war.0200 = {
	type = character_event
	theme = culture_change
	content_source = dlc_WtWSMS
	override_background = { reference = throne_room_scope }
	
	
	title = migration_situation_war.0200.t
	
	desc = migration_situation_war.0200.desc
	
	trigger = {
		has_dlc_feature = hybridize_culture #Safety
		BP_diverge_culture_during_migration_trigger = yes
		NOR = {
			culture = { has_cultural_tradition = tradition_nomadic_rulership }
			culture = { has_cultural_tradition = tradition_nomadic_rulership_cranial_deformation }
			culture = { has_cultural_tradition = tradition_migratory_culture }
		}
	}
	immediate = {}
	after = {}
	
	left_portrait = {
		character = ROOT
		animation = thinking
	}
	
	option = {
		name = migration_situation_war.become_migratory_culture
		
		custom_description = {
			text = new_culture_summary
			subject = root.culture
		}
		
		#Insert any culture name or name list shenanigans tooltips here
		
		if = {
			limit = { culture = { has_cultural_tradition = tradition_ruling_caste } }
			custom_tooltip = remove_ruling_cast_tradition
		}
		if = {
			limit = { culture = { has_cultural_tradition = tradition_cranial_deformation } }
			custom_tooltip = adopt_nomadic_rulership_cranial_deformation_tradition
		}
		else = { custom_tooltip = adopt_migratory_culture_tradition }
		
		hidden_effect = {
			create_divergent_culture = yes
			
			#Insert any culture name or name list shenanigans here
			
			scope:new_culture ?= {
				hidden_effect = {
					if = {
						limit = { NOT = { has_cultural_pillar = head_determination_herd } }
						set_culture_pillar = head_determination_herd
					}
				}
				#Ethos Determination
				random_list = {
					10 = {
						set_culture_pillar = ethos_bellicose
						modifier = { has_cultural_pillar = ethos_bellicose factor = 0 }
					}
					10 = {
						set_culture_pillar = ethos_stoic
						modifier = { has_cultural_pillar = ethos_stoic factor = 0 }
					}
					10 = {
						set_culture_pillar = ethos_bellicose
						modifier = { has_cultural_pillar = ethos_bellicose factor = 0 }
					}
					10 = {
						set_culture_pillar = ethos_bureaucratic
						modifier = { has_cultural_pillar = ethos_bureaucratic factor = 0 }
					}
					10 = {
						set_culture_pillar = ethos_spiritual
						modifier = { has_cultural_pillar = ethos_spiritual factor = 0 }
					}
					10 = {
						set_culture_pillar = ethos_courtly
						modifier = { has_cultural_pillar = ethos_courtly factor = 0 }
					}
					10 = {
						set_culture_pillar = ethos_egalitarian
						modifier = { has_cultural_pillar = ethos_egalitarian factor = 0 }
					}
					10 = {
						set_culture_pillar = ethos_communal
						modifier = { has_cultural_pillar = ethos_communal factor = 0 }
					}
				}
				#Tradition padding
				if = {
					limit = { has_cultural_tradition = tradition_ruling_caste }
					remove_culture_tradition = tradition_ruling_caste
				}
				if = {
					limit = { has_cultural_tradition = tradition_cranial_deformation }
					remove_culture_tradition = tradition_cranial_deformation
					add_culture_tradition = tradition_nomadic_rulership_cranial_deformation
				}
				else = { add_culture_tradition = tradition_migratory_culture }
			}
			
			recalculate_cultural_heads_of_type = domain
			recalculate_cultural_heads_of_type = herd
		}
		
		ai_chance = {
			base = 90
			modifier = { #Traditionalists Cultures say we won't do this
				factor = 0
				OR = {
					culture = { has_cultural_tradition = tradition_xenophobic }
					culture = { has_cultural_tradition = tradition_xenophobic_non_dlc }
					culture = { has_cultural_tradition = tradition_strong_kinship }
					culture = { has_cultural_tradition = tradition_staunch_traditionalists }
					culture = { has_cultural_tradition = tradition_mpo_northern_tribes }
					culture = { has_cultural_tradition = tradition_zealous_pagans }
					culture = { has_cultural_tradition = tradition_zealous_pagans_non_dlc }
					culture = { has_cultural_tradition = tradition_decentralized }
					culture = { has_cultural_tradition = tradition_decentralized_non_dlc }
				}
			}
			#Distance Ladder
			modifier = { #We're abutting our homeland
				factor = 0
				any_realm_county = { any_neighboring_county = { this.culture = root.culture } }
			}
			modifier = { #Our homeland is close
				factor = 0.1
				any_county = {
					squared_distance = { target = root.capital_county value < 10000 }
					this.culture = root.culture
				}
			}
			modifier = { #Our homeland is kinda close
				factor = 0.25
				any_county = {
					squared_distance = { target = root.capital_county value >= 10000 }
					squared_distance = { target = root.capital_county value < 50000 }
					this.culture = root.culture
				}
			}
			modifier = { #Our homeland is a bit of a walk
				factor = 0.50
				any_county = {
					squared_distance = { target = root.capital_county value >= 50000 }
					squared_distance = { target = root.capital_county value < 100000 }
					this.culture = root.culture
				}
			}
		}
	}
	option = {
		name = migration_situation_war.do_not_become_migratory_culture
		
		ai_chance = {
			base = 10
			modifier = { #Traditionalists Cultures say we keep to our ways
				factor = 1000
				OR = {
					culture = { has_cultural_tradition = tradition_xenophobic }
					culture = { has_cultural_tradition = tradition_xenophobic_non_dlc }
					culture = { has_cultural_tradition = tradition_strong_kinship }
					culture = { has_cultural_tradition = tradition_staunch_traditionalists }
					culture = { has_cultural_tradition = tradition_mpo_northern_tribes }
					culture = { has_cultural_tradition = tradition_zealous_pagans }
					culture = { has_cultural_tradition = tradition_zealous_pagans_non_dlc }
					culture = { has_cultural_tradition = tradition_decentralized }
					culture = { has_cultural_tradition = tradition_decentralized_non_dlc }
				}
			}
		}
	}
}
