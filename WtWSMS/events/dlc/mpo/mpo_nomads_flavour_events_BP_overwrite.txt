scripted_trigger wtwsms_nomd_conversion_event_trigger = {
	#In WtWSMS, let's avoid having religions spread too readily across the steppe
	OR = {
		current_year >= 800
		root.capital_county = {
			any_neighboring_county = {
				NOT = { this.faith = { has_doctrine_parameter = unreformed } }
			}
		}
	}
}
