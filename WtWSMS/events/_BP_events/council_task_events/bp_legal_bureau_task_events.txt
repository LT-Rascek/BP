legal_bureau_director_task.3001 = {
	hidden = yes

	trigger = {
		exists = cp:councillor_legal_bureau_director
		scope:councillor = cp:councillor_legal_bureau_director
		scope:councillor = {
			learning < high_skill_rating
		}
		scope:county = {
			NOT = { has_county_modifier = steward_settler_resistance_modifier }
		}
	}

	weight_multiplier = {
		base = 1
		compare_modifier = {
			target = scope:councillor
			value = learning
			multiplier = -0.2
			offset = inverted_high_skill_rating
		}
	}

	immediate = {
		cp:councillor_legal_bureau_director = {
			save_scope_as = active_councillor
		}
		set_variable = {
			name = had_legal_bureau_director_task_side_effect
			value = yes
			days = legal_bureau_director_task_side_effect_cooldown
		}

		send_interface_message = {
			type = msg_steward_task_bad
			title = legal_bureau_director_task.3001.notification
			desc = {
				triggered_desc = {
					trigger = {
						scope:active_councillor = { learning <= average_skill_level }
					}
					desc = task_learning_bad_unskilled_notification_tooltip
				}
				triggered_desc = {
					trigger = {
						scope:active_councillor = { learning > average_skill_level }
					}
					desc = task_learning_bad_skilled_notification_tooltip
				}
			}
			
			left_icon = scope:county
			right_icon = scope:councillor
			tooltip = task_promote_culture_notification_tooltip

			scope:county = {
				add_county_modifier = {
					modifier = steward_settler_resistance_modifier
					days = legal_bureau_director_task_modifier_duration
				}
			}
		}
	}
}

scripted_trigger legal_bureau_director_task_8001_vassal_trigger = {
	trigger_if = { #For directed efforts it has to be the target vassal
		limit = {
			scope:councillor = {
				is_performing_council_task = task_bestow_royal_favor
			}
		}
		this = scope:councillor.councillor_task_target 
	}
	trigger_else = {
		is_of_major_interest_to_root_trigger = yes
		is_imprisoned = no
		NOT = { has_trait = incapable }
		NOR = {
			has_opinion_modifier = {
				target = root
				modifier = chancellor_task_powerful_vassal_increased_opinion
			}
			has_opinion_modifier = {
				target = root
				modifier = chancellor_task_powerful_vassal_decreased_opinion
			}
		}
		NOT = { root.cp:councillor_chancellor ?= this }
	}
}

scripted_trigger legal_bureau_director_task_8101_defender_vassal = {
	is_ai = yes
	is_at_war_as_defender = yes
	save_temporary_scope_as = defender_vassal
	NOT = {
		root.cp:councillor_chancellor ?= this
	}
	trigger_if = { #For directed efforts one of the vassals has to be the target
		limit = {
			scope:councillor = {
				is_performing_council_task = task_bestow_royal_favor
			}
		}
		save_temporary_scope_as = vassal_1
	}
}

scripted_trigger legal_bureau_director_task_8101_attacker_vassal = {	
	is_ai = yes
	is_at_war_as_attacker = yes
	save_temporary_scope_as = attacker_vassal
	any_character_war = {
		primary_attacker = scope:attacker_vassal
		primary_defender = scope:defender_vassal
	}
	NOT = {
		root.cp:councillor_chancellor ?= this
	}
	trigger_if = { #For directed efforts one of the vassals has to be the target
		limit = {
			scope:councillor = {
				is_performing_council_task = task_bestow_royal_favor
			}
		}
		save_temporary_scope_as = vassal_2
	}
}

legal_bureau_director_task.8101 = {
	type = letter_event
	opening = {
		desc = legal_bureau_director_task.8101.opening
	}
	desc = legal_bureau_director_task.8101.desc
	sender = scope:active_councillor
	
	trigger = {
		NOT = { has_character_flag = had_msg_legal_bureau_director_task_8101 }
		OR = {
			AND = {
				exists = cp:councillor_chancellor
				scope:councillor = cp:councillor_chancellor
			}
			scope:councillor = {
				is_kurultai_trigger = yes
			}
		}
		is_at_war = no
		any_vassal = {
			legal_bureau_director_task_8101_defender_vassal = yes
		}
		any_vassal = {
			legal_bureau_director_task_8101_attacker_vassal = yes
		}
		trigger_if = { #For directed efforts one of the vassals has to be the target
			limit = {
				scope:councillor = {
					is_performing_council_task = task_bestow_royal_favor
				}
			}
			OR = {
				scope:target_character = scope:vassal_1
				scope:target_character = scope:vassal_2
			}
		}
		
		scope:councillor = {
			
			diplomacy > mediocre_skill_rating
		}
	}

	weight_multiplier = {
		base = 1
		compare_modifier = {
			target = scope:councillor
			value = diplomacy
			multiplier = 0.2
			offset = inverted_mediocre_skill_rating
		}
	}

	immediate = {
		add_character_flag = {
			flag = had_msg_legal_bureau_director_task_8101
			years = 5
		}
		scope:councillor = {
			save_scope_as = active_councillor
		}
		set_variable = {
			name = had_legal_bureau_director_task_side_effect
			value = yes
			days = legal_bureau_director_task_side_effect_cooldown
		}
		random_vassal = {
			limit = {
				legal_bureau_director_task_8101_defender_vassal = yes
				root = {
					any_vassal = {
						legal_bureau_director_task_8101_attacker_vassal = yes
					}
				}
			}
			save_scope_as = defender_vassal
		}
		random_vassal = {
			limit = {
				legal_bureau_director_task_8101_attacker_vassal = yes
			}
			save_scope_as = attacker_vassal
			random_character_war = {
				limit = {
					primary_attacker = scope:attacker_vassal
					primary_defender = scope:defender_vassal
				}
				save_scope_as = war
			}
		}
	}

	option = {
		name = legal_bureau_director_task.8101.a
		remove_treasury_or_gold = medium_treasury_or_gold_value
		scope:war = {
			end_war = white_peace
		}
		ai_chance = {
			base = 10
			modifier = {
				factor = 0
				short_term_treasury_or_gold <= medium_treasury_or_gold_value
			}
		}
	}

	option = {
		name = legal_bureau_director_task.8101.b
		add_prestige = medium_prestige_loss
		scope:war = {
			end_war = white_peace
		}
		ai_chance = {
			base = 100
		}
