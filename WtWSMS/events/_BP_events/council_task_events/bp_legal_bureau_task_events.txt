#Resistance to Settlers

legal_bureau_director_task.3001 = {
	hidden = yes

	trigger = {
		exists = cp:councillor_legal_bureau_director
		scope:councillor = cp:councillor_legal_bureau_director
		scope:councillor = {
			learning < high_skill_rating
		}
		scope:county = {
			NOT = { has_county_modifier = steward_settler_resistance_modifier }
		}
	}

	weight_multiplier = {
		base = 1
		compare_modifier = {
			target = scope:councillor
			value = learning
			multiplier = -0.2
			offset = inverted_high_skill_rating
		}
	}

	immediate = {
		cp:councillor_legal_bureau_director = {
			save_scope_as = active_councillor
		}
		set_variable = {
			name = had_legal_bureau_director_task_side_effect
			value = yes
			days = legal_bureau_director_task_side_effect_cooldown
		}

		send_interface_message = {
			type = msg_steward_task_bad
			title = legal_bureau_director_task.3001.notification
			desc = {
				triggered_desc = {
					trigger = {
						scope:active_councillor = { learning <= average_skill_level }
					}
					desc = task_learning_bad_unskilled_notification_tooltip
				}
				triggered_desc = {
					trigger = {
						scope:active_councillor = { learning > average_skill_level }
					}
					desc = task_learning_bad_skilled_notification_tooltip
				}
			}
			
			left_icon = scope:county
			right_icon = scope:councillor
			tooltip = task_promote_culture_notification_tooltip

			scope:county = {
				add_county_modifier = {
					modifier = steward_settler_resistance_modifier
					days = legal_bureau_director_task_modifier_duration
				}
			}
		}
	}
}

scripted_trigger legal_bureau_director_task_9003_baron_trigger = {
	NOR = {
		has_opinion_modifier = {
			target = root
			modifier = marshal_task_increase_control_baron_increase_opinion
		}
		has_opinion_modifier = {
			target = root
			modifier = marshal_task_increase_control_baron_decrease_opinion
		}
	}
	this != scope:councillor
}

scripted_trigger legal_bureau_director_task_9003_county_trigger = {
	any_in_de_facto_hierarchy = {
		tier = tier_barony
		exists = holder
		holder != ROOT
		holder = {
			legal_bureau_director_task_9003_baron_trigger = yes
		}
	}
}
