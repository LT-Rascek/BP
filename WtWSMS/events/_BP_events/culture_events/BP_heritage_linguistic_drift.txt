scripted_trigger ruler_is_valid_to_hear_about_culture_drift_trigger = { #Scope is ruler in question
	is_ai = no #Otherwise hangs
	OR = {
		#Same Language/Heritage always tells your
		this.culture = { $HAS_SAME_LANGUAGE_HERITAGE_TRIGGER$ = scope:drift_culture }
		#Adjacent to your realm
		any_realm_county = { any_neighboring_county = { this.culture = scope:drift_culture } }
		#Is a small but close culture
		AND = {
			scope:drift_culture = { any_culture_county = { count <= 5 } }
			scope:drift_culture = { any_culture_county = { squared_distance < 100000 } }
		}
		#Is a medium culture
		AND = {
			scope:drift_culture = { any_culture_county = { count <= 10 } }
			scope:drift_culture = { any_culture_county = { squared_distance < 500000 } }
		}
		#Is a large culture
		AND = {
			scope:drift_culture = { any_culture_county = { count <= 15 } }
			scope:drift_culture = { any_culture_county = { squared_distance < 1000000 } }
		}
		#Is a huge culture
		AND = {
			scope:drift_culture = { any_culture_county = { count <= 20 } }
			scope:drift_culture = { any_culture_county = { squared_distance < 5000000 } }
		}
		#Gargantuan cultures you always hear about
		scope:drift_culture = { any_culture_county = { count >= 21 } }
	}
}

#Mostly we want this case to be where cultures are "well packed (i.e., small)" to allow differences to build
scripted_trigger is_culture_mostly_in_linguistic_region_trigger = {
	any_culture_county = {
		percent >= 0.50
		title_province = { geographical_region = $REGION$ }
	}
}

scripted_trigger a_culture_parent_is_a_germanic_culture_trigger = {
	any_parent_culture = { has_cultural_parameter = heritage_group_germanic }
}

scripted_trigger a_culture_parent_speaks_a_semitic_language_trigger = {
	any_parent_culture = { has_cultural_parameter = language_branch_semitic }
}

	any_culture_county = { any_neighboring_county = { this.culture = { has_cultural_pillar = $HERITAGE$ } } }

scripted_effect teach_chars_new_language = {
	save_temporary_scope_as = temp_language_culture
	every_living_character = {
		limit = { this.culture = scope:temp_language_culture }
		if = {
			limit = { NOT = { knows_language_of_culture = scope:temp_language_culture } }
			learn_language_of_culture = scope:temp_language_culture
		}
	}
}

scripted_effect shift_language_proto_ibero_romance_to_children_effect = {
	scope:drift_culture = {
		if = {
			limit = { is_a_iberian_romance_hybrid_culture_trigger = yes }
			if = {
				limit = { a_culture_parent_is_a_germanic_culture_trigger = yes }
				set_culture_pillar = language_iberian
				ccu_culture_created_or_changed_effect = yes
			}
			else_if = {
				limit = { a_culture_parent_speaks_a_semitic_language_trigger = yes }
				set_culture_pillar = language_mozarabic
				ccu_culture_created_or_changed_effect = yes
			}
			else = { #FALLBACK
				set_culture_pillar = language_iberian_romance
				ccu_culture_created_or_changed_effect = yes
			}
		}
		else = {
			if = {
				limit = { is_culture_mostly_in_linguistic_region_trigger = { REGION = custom_doc_region } }
				set_culture_pillar = language_doc_romance
				ccu_culture_created_or_changed_effect = yes
			}
			else = {
				set_culture_pillar = language_iberian_romance
				ccu_culture_created_or_changed_effect = yes
			}
		}
	}
	
}

#Main Language Shift Effect
scripted_effect shift_language_effect = {
	scope:drift_culture = {
		#First find all applicable minorities
		find_same_language_minorities = yes
		
		#Then apply the new language effect
		#NB: if/else-if prevents possible chain reaction

# Make list of this culture and all cultures that exist only as minorities under them
scripted_effect find_same_language_minorities = {
	# Start with scope:drift_culture
	add_to_list = all_drift_cultures
	
	# No counties, same language
	every_culture_global = {
		limit = {
			has_same_culture_language = scope:drift_culture
			any_culture_county = { count = 0 }
		}
		add_to_temporary_list = check_culture_minorities
	}
	# Must have 50% or more of their minorities under the shift culture
	every_in_list = {
		list = check_culture_minorities
		save_scope_as = check_minority_culture
		
		# If most are under the shift culture, save them to the persistent list
		if = {
			limit = {
				this.number_of_drift_culture_counties_with_minority > 0
				this.number_of_drift_culture_counties_with_minority >= this.half_number_of_counties_with_minority
			}
			add_to_list = all_drift_cultures
		}
	}
}

	scope:drift_culture = {
		if = {
			limit = { the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_hispanic } }

		}

			limit = { the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_gallic } }
			
			if = { #If ruled by Franks or a Frankish hybrid, becomes Frankish
				limit = {
					OR = {
						any_parent_culture = { this = culture:old_frankish }
						any_culture_county = { percent >= 0.50 this.holder.top_liege.culture = { any_parent_culture = { this = culture:old_frankish } } }
					}
				}
				set_culture_pillar = heritage_frankish
				ccu_culture_created_or_changed_effect = yes
			}
			else = { #Otherwise, Gallic Romance
				set_culture_pillar = heritage_gallic_romance
				ccu_culture_created_or_changed_effect = yes
			}
		}
		else_if = {
			limit = { the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_british } }
			set_culture_pillar = heritage_romano_british
			ccu_culture_created_or_changed_effect = yes
		}
		else_if = {
			limit = { the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_aramean } }
			set_culture_pillar = heritage_romano_aramean
			ccu_culture_created_or_changed_effect = yes
		}
		else_if = {
			limit = { the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_african } }
			set_culture_pillar = heritage_romano_african
			ccu_culture_created_or_changed_effect = yes
		}
		else_if = {
			limit = {
				OR = {
					the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_dacian }
					the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_pannonian }
					the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_thracian }
					the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_noric }
				}
			}
			set_culture_pillar = heritage_vlach
			ccu_culture_created_or_changed_effect = yes
		}
		else_if = {
			limit = {
				OR = {
					the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_illyrian }
					the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:romano_raetian }
					the_culture_or_parent_or_above_is_trigger = { CULTURE = culture:sardinian }

			ccu_culture_created_or_changed_effect = yes
		}

	}
}

			limit = {
				OR = {
					has_cultural_pillar = heritage_suebic
					is_central_germanic_saxon_trigger = yes
				}
			}
			germanic_shift_by_local_cultures_effect = { DEFAULT_MEDIEVAL_HERITAGE = heritage_central_germanic }
		}
		else_if = { #East Germanics...
			limit = {
				OR = {
					has_cultural_pillar = heritage_gothic
					has_cultural_pillar = heritage_przeworsk
				}
			}
			germanic_shift_by_local_cultures_effect = { DEFAULT_MEDIEVAL_HERITAGE = heritage_east_germanic }
		}
		else = { debug_log = "UNHANDLED CULTURE IN bp_heritage_linguistic_drift EVENT SPACE: HERITAGE (GERMANIC=>CHILDREN)" }
	}
}

scripted_effect shift_special_heritage_to_children_effect = {
	scope:drift_culture = {
		if = {
			limit = { is_carpodaci_trigger = yes }
			calc_culture_dominant_faith = { save_temporary_scope_as = drift_culture_dominant_religion }
			if = {
				limit = { scope:drift_culture_dominant_religion = religion:christianity_religion }
				set_culture_pillar = heritage_vlach
			}
			else = {
				set_culture_pillar = heritage_albanian
			}
			ccu_culture_created_or_changed_effect = yes
		}
	}
}


scripted_effect shift_heritage_effect = {
	shift_special_heritage_to_children_effect = yes
	shift_latin_heritage_to_children_effect = yes
	shift_germanic_heritage_to_children_effect = yes
}

bp_heritage_linguistic_drift.0101 = {
	scope = none
	hidden = yes
	content_source = dlc_WtWSMS
	
	immediate = {
		debug_set_drift_culture = yes
		#If the culture head exists, inform him/her first
		if = {
			limit = { exists = scope:drift_culture.culture_head }
			if = {
				# Certain cultures should get to decide their heritage
				limit = {
					scope:drift_culture = {
						culture_head = { is_ai = no }
						is_carpodaci_trigger = yes
					}
				}
				scope:drift_culture.culture_head = { trigger_event = { id = bp_heritage_linguistic_drift.0104 } }
			}	
			else = {
				scope:drift_culture.culture_head = { trigger_event = { id = bp_heritage_linguistic_drift.0102 } }
			}
		}
		#Otherwise, directly change the culture and tell everyone
		else = {
			shift_heritage_effect = yes
			every_ruler = { #Informational event
				limit = {
					ruler_is_valid_to_hear_about_culture_drift_trigger = {
						HAS_SAME_LANGUAGE_HERITAGE_TRIGGER = has_same_culture_heritage
					}
				}
				trigger_event = { id = bp_heritage_linguistic_drift.0103 }
			}
		}
	}
}

bp_heritage_linguistic_drift.0103 = {
	type = character_event
	title = bp_heritage_linguistic_drift.0103.t
	desc = {
		first_valid = {
			#Latin to Romance
			triggered_desc = {
				trigger = { scope:drift_culture = { has_cultural_pillar = heritage_latin } }
				desc = bp_heritage_linguistic_drift.0103.latin.desc
			}
			#Default
			desc = bp_heritage_linguistic_drift.0103.desc
		}
	}
	theme = culture_change
	override_background = { reference = BP_culture_drift_city }
	content_source = dlc_WtWSMS
	left_portrait = scope:bg_override_char
	lower_right_portrait = scope:my_portrait
	
	immediate = {
		if = {
			limit = { exists = scope:drift_culture.culture_head }
			scope:drift_culture.culture_head = { save_scope_as = bg_override_char }
		}
		else_if = {
			limit = {
				any_ruler = {
					culture = scope:drift_culture
					NOR = { #Keep it to chars functionally on the map please
						government_has_flag = cannot_be_vassal_or_liege
						government_has_flag = government_is_herder
					}
				}
			}
			random_ruler = {
				limit = { culture = scope:drift_culture }
				save_scope_as = bg_override_char
			}
		}
		# Add your portrait if your culture matches
		if = {
			limit = { root.culture = scope:drift_culture }
			save_scope_as = my_portrait
		}
	}
	
	#For other rulers of the culture
	option = {
		name = bp_heritage_linguistic_drift.0103.accept_same_culture
		flavor = bp_heritage_linguistic_drift.0103.accept_same_culture.tooltip
		trigger = { root.culture = scope:drift_culture }
	}
	#For other rulers not of the culture
	option = {
		name = bp_heritage_linguistic_drift.0103.accept_different_culture
		flavor = bp_heritage_linguistic_drift.0103.accept_different_culture.tooltip
		trigger = { NOT = { root.culture = scope:drift_culture } }
		fallback = yes
	}
}
