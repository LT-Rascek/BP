##### Changed from Vanilla: Not Available to Despoilers
tgp_japan_legacy_track = {
	is_shown = {
		has_dlc_feature = all_under_heaven
		OR = {
			has_game_rule = unrestricted_dynasty_legacies_all
			AND = {
				has_game_rule = unrestricted_dynasty_legacies_player_only
				is_ai = no
			}
			government_has_flag = government_is_japan_administrative
			government_has_flag = government_is_japan_feudal
			dynasty = {
				has_dynasty_perk = tgp_japan_legacy_1
			}
		}
		##### Changed from Vanilla vvvvv
		NOT = { dynasty = { has_dynasty_perk = BP_despoilation_legacy_1 } }
		##### Changed from Vanilla ^^^^^
	}
}
