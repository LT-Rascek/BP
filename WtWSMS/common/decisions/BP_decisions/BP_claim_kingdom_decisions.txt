##### These are various decisions that cover weird edge cases between the CK3 mechanics and history
##### Basically, should be a cheaper, simpler way of getting certain kingdoms that would exist historically

#The Aures Kingdom in the mountains of Africa
claim_k_aures_decision = {
	picture = {
		reference = "gfx/interface/illustrations/holding_types/fp2_castle_iberian.dds"
	}
	decision_group_type = major
	
	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 60
		kingdom = 0
		empire = 0
		hegemony = 0
	}
	
	desc = claim_k_aures_decision_desc
	selection_tooltip = claim_k_aures_decision_tooltip
	
	is_shown = {
		highest_held_title_tier = tier_duchy
		has_title = title:d_auresia
		NOT = { has_global_variable = claimed_k_aures }
	}
	
	is_valid = {
		OR = {
			#Has enough political clout
			prestige_level >= 2
			piety_level >= 2
		}
		has_title = title:d_auresia
		#Have to be in the Aures Mountains
		OR = {
			custom_description = {
				text = capital_is_bilizma
				root.capital_county = title:c_bilizma
			}
			custom_description = {
				text = capital_is_biskra
				root.capital_county = title:c_biskra
			}
		}
	}
	
	is_valid_showing_failures_only = {
		is_available_at_peace_adult = yes
		is_imprisoned = no
		is_adult = yes
	}
	
	cost = {
		treasury = {
			value = 0
			if = {
				limit = { has_treasury = yes }
				add = 25
			}
		}
		gold = {
			value = 0
			if = {
				limit = { has_treasury = no }
				add = 25
			}
		}
		prestige = 50
		piety = 50
	}
	
	effect = {
		#Done as event to prevent RC formation event from firing
		show_as_tooltip = { form_k_aures_effect = yes }
		hidden_effect = {
			#Can only do it once
			set_global_variable = {
				name = claimed_k_aures
				value = yes
			}
			
			root = { save_scope_as = the_k_aures_former }
			
			root = { trigger_event = { id = BP_claim_kingdom_decisions.0020 } }
		}
	}
	
	ai_potential = { always = yes }
	
	ai_will_do = { base = 100 }
}
