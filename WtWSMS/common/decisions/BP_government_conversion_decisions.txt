convert_to_high_tribal_decision = {
	picture = {
		reference = "gfx/interface/illustrations/character_view/fp1_throneroom_nontribal.dds"
	}
	decision_group_type = major
	
	title = {
		first_valid = {
			triggered_desc = {
				trigger = {
					government_has_flag = is_a_transitory_roman_government
					this.primary_title.tier <= tier_duchy
				}
				desc = devolve_to_high_tribal_decision
			}
			desc = convert_to_high_tribal_decision
		}
	}
	
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					government_has_flag = is_a_transitory_roman_government
					this.primary_title.tier <= tier_duchy
				}
				desc = devolve_to_high_tribal_decision_desc
			}
			desc = convert_to_high_tribal_decision_desc
		}
	}
	
	ai_check_interval_by_tier = {
		barony = 0
		county = 72
		duchy = 60
		kingdom = 48
		empire = 36
		hegemony = 36
	}
	
	is_shown = {
		top_liege = this
		is_landed = yes
		NOT = { government_has_flag = government_is_nomadic }
		NOT = { government_has_flag = government_is_herder }
		OR = {
			AND = {
				government_has_flag = government_is_tribal
				NOT = { government_has_flag = government_is_high_tribal }
			}
			AND = {
				government_has_flag = is_a_transitory_roman_government
				this.primary_title.tier <= tier_duchy
			}
		}
	}
	
	is_valid = {
		trigger_if = {
			limit = {
				government_has_flag = government_is_tribal
				NOT = { government_has_flag = government_is_high_tribal }
			}
			has_realm_law = tribal_authority_3
			prestige_level >= 2
			capital_province.county = { development_level >= convert_to_high_tribal_development_requirement }
			
			culture = { has_tribal_innovations_for_high_tribal_trigger = yes }
		}
		trigger_else_if = {
			limit = {
				AND = {
					government_has_flag = is_a_transitory_roman_government
					this.primary_title.tier <= tier_duchy
				}
			}
			#Romans are under significant constraints
			trigger_if = {
				limit = {
					OR = {
						culture = { has_cultural_parameter = greco_roman_social_tradition }
						culture = { has_cultural_parameter = can_restore_roman_empire }
					}
				}
				#We're too small, too poor, or too obscure to keep up the government
				OR = {
					custom_tooltip = {
						text = realm_size_less_than_or_equal_to_1
						realm_size <= 1
					}
					this.dynasty.dynasty_prestige_level < 1
				}
				NOT = { has_trait = legendary_leader }
			}
			#Non-Romans can pretty much do this whenever the realm is fairly contracted
			trigger_else = {
				custom_tooltip = {
					text = realm_size_less_than_or_equal_to_3
					realm_size <= 3
				}
			}
		}
		trigger_else = {} #Error suppression
	}
	
	cost = {
		prestige = {
			value = 0
			if = {
				limit = {
					government_has_flag = government_is_tribal
					NOT = { government_has_flag = government_is_high_tribal }
				}
				add = 250
			}
			else_if = {
				limit = {
					government_has_flag = is_a_transitory_roman_government
					this.primary_title.tier <= tier_duchy
				}
				add = 50
			}
		}
	}
	
	effect = {
		trigger_event = BP_government_conversion.0001
		save_scope_as = actor
		show_as_tooltip = { change_government = high_tribal_government }
	}
	
	ai_potential = {
		trigger_if = {
			limit = {
				government_has_flag = is_a_transitory_roman_government
				this.primary_title.tier <= tier_duchy
			}
			#Masties would _never_ do this
			NOT = { this = character:36141 }
			#Ambrosius would _never_ do this
			NOT = { this = character:190007 }
			#Arbogastes would _never_ do this
			NOT = { this = character:144671 }
			#Sala Colonia is Special
			this.primary_title = title:c_gharb
			#Saberia is Special
			this.primary_title = title:d_montes_aregenses
			#Ceuta is Special
			this.primary_title = title:d_tangiers
		}
	}
	
	ai_will_do = {
		base = 0
		#Tribals will always do this
		modifier = {
			add = 1000
			government_has_flag = government_is_tribal
		}
		#AI will do when they are basically tribal already
		modifier = {
			add = 100
			AND = {
				government_has_flag = is_a_transitory_roman_government
				this.primary_title.tier <= tier_duchy
			}
			NOR = {
				culture = { has_cultural_parameter = greco_roman_social_tradition }
				culture = { has_cultural_parameter = can_restore_roman_empire }
			}
			OR = {
				root.faith = { has_doctrine_parameter = unreformed }
				culture = {
					num_discovered_innovations_in_era = {
						target = culture_era_classical_antiquity
						value < 3
					}
				}
			}
		}
		#AI will do if they have a big bad, non-Roman next to them
		#If they are three times our strength, we just immediately land troops
		modifier = {
			add = 100
			AND = {
				government_has_flag = is_a_transitory_roman_government
				this.primary_title.tier <= tier_duchy
			}
			any_neighboring_and_across_water_realm_owner = {
				NOT = { government_has_flag = is_a_roman_government }
				root.max_military_strength < this.one_third_military_strength
			}
		}
	}
}

convert_to_high_tribal_as_vassal_decision = {
	picture = {
		reference = "gfx/interface/illustrations/character_view/fp1_throneroom_nontribal.dds"
	}
	decision_group_type = major
	desc = convert_to_high_tribal_as_vassal_decision_desc
	
	ai_check_interval_by_tier = {
		barony = 0
		county = 72
		duchy = 60
		kingdom = 48
		empire = 36
		hegemony = 36
	}
	
	is_shown = {
		government_has_flag = government_is_tribal
		NOT = { government_has_flag = government_is_nomadic }
		NOT = { government_has_flag = government_is_high_tribal }
		top_liege != this
		liege = {
			OR = {
				government_has_flag = government_is_centralized
				AND = {
					has_a_feudal_type_of_government_trigger = yes
					NOT = { government_has_flag = government_is_prefeudal }
				}
			}
		}
	}
	
	is_valid = {
		OR = {
			has_realm_law = tribal_authority_1
			has_realm_law = tribal_authority_2
			has_realm_law = tribal_authority_3
		}
		custom_description = {
			text = recent_conversion_to_tribalism
			NOT = { has_character_flag = just_converted_to_tribalism }
		}
		culture = { has_innovation = innovation_writing }
	}
	
	cost = { prestige = 150 }
	
	effect = {
		trigger_event = BP_government_conversion.0011
		save_scope_as = actor
		show_as_tooltip = { change_government = high_tribal_government }
	}
	
	ai_potential = { always = yes }
	
	ai_will_do = { base = 100 }
}

convert_to_roman_government_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/ep3_cerimonial_decision.dds"
	}
	decision_group_type = major
	desc = convert_to_roman_government_decision_desc
	
	ai_check_interval_by_tier = {
		barony = 0
		county = 72
		duchy = 60
		kingdom = 48
		empire = 36
		hegemony = 36
	}
	
	is_shown = {
		trigger_if = {
			limit = { this.dynasty ?= { has_dynasty_perk = BP_roman_heritage_legacy_5 } }
			NOR = {
				#Not already main Roman Government
				government_has_flag = government_is_roman_dynastic
				government_has_flag = government_is_roman_administrative
				#Can't be Chinese Governments
				government_has_flag = government_is_celestial
				government_has_flag = government_is_mandala
				government_has_flag = government_is_steppe_admin
				government_has_flag = government_is_meritocratic
			}
			#Have to be some sort of applicable government
			OR = {
				government_has_flag = government_is_feudal
				government_has_flag = government_is_clan
				government_has_flag = government_is_iranian_dynastic
				government_has_flag = government_is_rajamandala
				government_has_flag = government_is_administrative
				government_has_flag = is_a_transitory_roman_government
			}
		}
		trigger_else = { government_has_flag = is_a_transitory_roman_government }
	}
	
	is_valid = {
		custom_description = {
			text = faith_is_reformed
			subject = root
			faith = { NOT = { has_doctrine_parameter = unreformed } }
		}
		top_liege = this #Must be independent
		highest_held_title_tier >= tier_kingdom #Must be a kingdom or empire or hegemony
		prestige_level >= 4
		capital_province.county = { development_level >= convert_to_roman_government_development_requirement }
		
		#Tribal Innovations
		custom_description = {
			text = text_has_all_tribal_innovations
			subject = root.culture
			culture = { has_all_necessary_tribal_inovations_trigger = yes }
		}
		
		#Classical Antiquity Innovations
		custom_description = {
			text = text_has_all_classical_antiquity_innovations
			subject = root.culture
			culture = { has_all_necessary_classical_antiquity_innovations_trigger = yes }
		}
		
		culture = {
			has_innovation = innovation_aristocracy
			has_innovation = innovation_hereditary_rule
		}
	}
	
	cost = { prestige = 250 }
	
	effect = {
		trigger_event = BP_government_conversion.0021
		save_scope_as = actor
		show_as_tooltip = { convert_to_roman_government_effect = yes }
	}
	
	ai_potential = { always = yes }
	
	ai_will_do = { base = 100 }
}

convert_to_feudalism_decision = {
	picture = {
		reference = "gfx/interface/illustrations/loading_screens/council.dds"
	}
	decision_group_type = major
	desc = convert_to_feudalism_decision_desc
	
	ai_check_interval_by_tier = {
		barony = 0
		county = 72
		duchy = 60
		kingdom = 48
		empire = 36
		hegemony = 36
	}
	
	is_shown = {
		OR = {
			trigger_if = {
				limit = { dynasty ?= { has_dynasty_perk = BP_despoilation_legacy_5 } }
				government_has_flag = government_is_tribal
			}
			government_has_flag = government_is_prefeudal
			government_has_flag = government_is_high_tribal
			government_has_flag = government_is_subroman_administrative
		}
		#TODO: Fix to match the other govs Gov't decisions so we don't get a case where a player has no conversion decision
		#OR = {
		#	culture = { has_cultural_parameter = right_to_rule_feudal }
		#	faith = { has_doctrine_parameter = right_to_rule_feudal }
		#}
	}
	
	is_valid = {
		capital_province.county = { development_level >= convert_to_feudalism_development_requirement }
		custom_description = {
			text = faith_is_reformed
			subject = root
			faith = { NOT = { has_doctrine_parameter = unreformed } }
		}
		custom_tooltip = {
			text = has_maxed_out_authority
			has_realm_law_flag = max_authority_level
		}
		trigger_if = {
			limit = { dynasty ?= { has_dynasty_perk = BP_despoilation_legacy_5 } }
			prestige_level >= 4
		}
		trigger_else = {
			trigger_if = {
				limit = { government_has_flag = government_is_subroman_administrative }
				top_liege = root
				OR = {
					has_realm_law = subroman_administrative_military_centralization_top_liege_law_4
					has_realm_law = subroman_administrative_military_centralization_top_liege_law_5
				}
				OR = {
					has_realm_law = subroman_administrative_taxation_top_liege_law_3
					has_realm_law = subroman_administrative_taxation_top_liege_law_4
				}
				OR = {
					has_realm_law = subroman_administrative_aristocracy_law_3
					has_realm_law = subroman_administrative_aristocracy_law_4
				}
			}
			prestige_level >= 2
			#Tribal Innovations
			custom_description = {
				text = text_has_all_tribal_innovations
				subject = root.culture
				culture = { has_all_necessary_tribal_inovations_trigger = yes }
			}
			
			#Classical Antiquity Innovations
			custom_description = {
				text = text_has_all_classical_antiquity_innovations
				subject = root.culture
				culture = { has_all_necessary_classical_antiquity_innovations_trigger = yes }
			}
			
			#Implicit for High Tribal Organization
			trigger_if = {
				limit = { NOT = { government_has_flag = government_is_high_tribal } }
				culture = {
					has_innovation = innovation_aristocracy
					has_innovation = innovation_hereditary_rule
				}
			}
		}
	}
	
	cost = { prestige = 250 }
	
	effect = {
		trigger_event = BP_government_conversion.0031
		save_scope_as = actor
		show_as_tooltip = {
			change_government = feudal_government
			custom_tooltip = {
				text = convert_holdings_text
				convert_holdings_effect = yes
			}
		}
	}
	
	ai_potential = { always = yes }
	
	ai_will_do = {
		base = 0
		#High Tribal/Prefeudal will always do this
		modifier = {
			add = 100
			OR = {
				government_has_flag = government_is_prefeudal
				government_has_flag = government_is_high_tribal
			}
		}
		#Sub-Roman AI will only do it when their government has degraded enough
		modifier = {
			government_has_flag = government_is_subroman_administrative
			has_realm_law = subroman_administrative_military_centralization_top_liege_law_5
			has_realm_law = subroman_administrative_taxation_top_liege_law_4
			OR = {
				has_realm_law = subroman_administrative_aristocracy_law_3
				has_realm_law = subroman_administrative_aristocracy_law_4
			}
		}
	}
}
