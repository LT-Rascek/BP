#All these are to be done _AFTER_ every other on action for priority purposes.

on_game_start_after_lobby = {
	on_actions = {
		establish_roman_premierships
		temp_patch_of_bga_bugs
	}
	events = {}
}

establish_roman_premierships = {
	trigger = { BP_has_dlc_necessary_for_roman_admin_governments_trigger = yes }
	effect = {
		every_ruler = {
			limit = { bp_has_access_to_roman_ministry_trigger = yes }
			establish_premiership_if_necessary_effect = yes
		}
	}
}

temp_patch_of_bga_bugs = {
	trigger = { has_dlc_feature = by_god_alone }
	effect = {
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
		}
		title:d_et_kartli = {
			change_title_holder = {
				holder = title:d_et_kartli.previous_holder
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		title:d_et_kartli.holder = { change_government = ecclesiastical_government }
	}
}
