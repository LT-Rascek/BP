free_diarchy_law = {
	law_group_type = roman_diarchy_law
	index = 0

	flag = free_diarchy_law
	
	can_have = {
		bp_has_access_to_roman_ministry_trigger = yes
		has_tgp_dlc_trigger = yes # Needs ministry titles to exist
	}

	can_pass = {
		custom_tooltip = {
			text = admin_law_cooldown_desc
			NOT = { has_variable = admin_law_cooldown }
		}
		#Other Diarchy Requirements
		trigger_if = {
			limit = { has_realm_law = roman_chancellor_law }
			custom_tooltip = {
				text = current_diarch_must_be_weak_or_loyal
				any_vassal = {
					any_held_title = { has_variable = is_supreme_roman_ministry_title }
					save_temporary_scope_as = the_diarch
				}
				NOT = { scope:the_diarch ?= { has_hook = root } }
				OR = {
					root.diarch ?= { diarch_loyalty >= 75 }
					root = { diarchy_swing < 5 }
				}
			}
		}
		trigger_if = {
			limit = { has_realm_law = roman_marshal_law }
			custom_tooltip = {
				text = current_diarch_must_be_weak_or_loyal
				any_vassal = {
					any_held_title = { has_variable = is_supreme_roman_military_title }
					save_temporary_scope_as = the_diarch
				}
				NOT = { scope:the_diarch ?= { has_hook = root } }
				OR = {
					root.diarch ?= { diarch_loyalty >= 75 }
					root = { diarchy_swing < 5 }
				}
			}
		}
	}

	on_pass = {
		## Set cooldown
		diarch = { depose_diarch = yes }
		set_variable = {
			name = admin_law_cooldown
			years = @admin_law_cooldown_value
		}
	}

	#Roman Bureaucracy does not like a free type of diarchy
	modifier = {
		roman_administrative_government_vassal_opinion = -10
		roman_administrative_government_tax_contribution_mult = -0.25
		roman_administrative_government_levy_contribution_mult = -0.25
		enemy_hostile_scheme_success_chance_max_add = 5
		monthly_influence_mult = -0.1
		close_relative_opinion = 10
	}

	ai_will_do = {
		value = 1
	}
}

	law_group_type = non_roman_minority_rule_law
	index = 0

	can_keep = {
		bp_can_have_roman_minority_law_trigger = yes
		#Lock into top liege's law
		trigger_if = {
			limit = { top_liege != this }
			liege = { has_realm_law = non_roman_minority_rule_law_0 }
		}
	}
	
	modifier = {
		cultural_head_acceptance_gain_mult = -0.50
		same_culture_opinion = 20
		different_culture_opinion = -20
	}
	flag = close_family_gets_appointment_exception
	flag = only_same_culture_acclamation_succession
	flag = only_same_culture_appointment_succession
	flag = will_not_convert_to_local_culture
	flag = roman_characters_can_be_minorities
	
	can_pass = {
		#Only Top liege
		top_liege = this
		#Only adjacent
		has_realm_law = non_roman_minority_rule_law_1
		#Not on cooldown
		custom_description = {
			subject = root
			text = "has_non_roman_minority_rule_law_cooldown"
			NOT = { has_variable = non_roman_minority_rule_law_cooldown }
		}
	}
	
	#Shifting laws should please Romans/anger minorities unless fickle/eccentric (opposite of stubborn)
	pass_cost = {
		prestige = {
			if = { #Only the top liege pays for this
				limit = { top_liege = this }
				add = change_non_roman_minority_rule_law_basic_cost
			}
		}
		influence = {
			if = { #Only the top liege pays for this
				limit = {
					top_liege = this
					government_has_flag = government_has_influence
				}
				add = change_non_roman_minority_rule_law_basic_cost
			}
		}
	}
	on_pass = {
		set_variable = {
			name = non_roman_minority_rule_law_cooldown
			years = @non_roman_minority_rule_law_cooldown
		}
		#Apply opinions kingdom-wide when the ruler applies a new law
		if = {
			limit = {
				top_liege = this
				has_realm_law = non_roman_minority_rule_law_1
			}
			custom_description = {
				text = all_roman_vassal_disapprove
				subject = root
				value = change_non_roman_minority_rule_law_opinion_value_negative
				every_vassal_or_below = {
					limit = { culture = { has_cultural_parameter = greco_roman_social_tradition } }
					add_opinion = {
						modifier = reduced_roman_priveleges_roman
						target = root
					}
				}
				every_courtier = {
					limit = { culture = { has_cultural_parameter = greco_roman_social_tradition } }
					add_opinion = {
						modifier = reduced_roman_priveleges_roman
						target = root
					}
				}
			}
			custom_description = {
				text = all_vassals_your_culture_or_migratory_approve
				subject = root
				value = change_non_roman_minority_rule_law_opinion_value
				every_vassal_or_below = {
					limit = {
						OR = {
							culture = root.culture
							culture = { has_cultural_tradition = tradition_elite_rulership }
						}
					}
					add_opinion = {
						modifier = reduced_roman_priveleges_minority
						target = root
					}
				}
				every_courtier = {
					limit = {
						OR = {
							culture = root.culture
							culture = { has_cultural_tradition = tradition_elite_rulership }
						}
					}
					add_opinion = {
						modifier = reduced_roman_priveleges_minority
						target = root
					}
				}
			}
		}
		#Keep the laws united
		hidden_effect = {
			every_vassal_or_below = { add_realm_law_skip_effects = non_roman_minority_rule_law_0 }
		}
	}
	ai_will_do = {
		if = { #Vandals *want* to oppress Romans
			limit = { culture = culture:vandalic }
			value = 1000
		}
		#Ethos
		if = { #Communal Ethos like themselves
			limit = { culture = { has_cultural_pillar = ethos_communal } }
			add = 50
		}
		if = { #Egalitarians are disinclined to oppress
			limit = { culture = { has_cultural_pillar = ethos_egalitarian } }
			add = -50
		}
		#Bonuses
		if = { #WtWSMS: For the Love of the Game
			limit = {
				culture = { has_cultural_tradition = tradition_punic_traditions }
				culture = { has_cultural_tradition = tradition_persian_traditions }
				culture = { has_cultural_tradition = tradition_persian_traditions_non_dlc }
			}
			value = 200
		}
		if = { #WtWSMS Traditionalists/Anti-Romans
			limit = {
				culture = { has_cultural_tradition = tradition_marauders }
				culture = { has_cultural_tradition = tradition_marauders_non_dlc }
				culture = { has_cultural_tradition = tradition_xenophobic }

				culture = { has_cultural_tradition = tradition_autochthon_identity }

				culture = { has_cultural_tradition = tradition_armenian_traditions_non_dlc }
				culture = { has_cultural_tradition = tradition_hebrew_traditions }
				culture = { has_cultural_tradition = tradition_hebrew_traditions_non_dlc }
				culture = { has_cultural_tradition = tradition_egyptian_traditions }

				culture = { has_cultural_tradition = tradition_caucasian_mountain_homes }
				culture = { has_cultural_tradition = tradition_caucasian_mountain_homes_non_dlc }
				culture = { has_cultural_tradition = tradition_early_slavic_traditions }
			}
			value = 75
		}
		if = { #Staunch Traditionalists don't like Romans
			limit = { culture = { has_cultural_parameter = harder_to_hybridize } }
			value = 75
		}
		if = { #Strong believers keep to themselves (can't use the calc_faith trigger?)

		}
		if = { #Jew's don't care to be Roman
