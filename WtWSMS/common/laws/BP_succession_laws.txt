#Order Laws
roman_acclamation_succession_law = {
	law_group_type = succession_order_laws
	index = 35

	can_keep = {
		can_have_roman_acclamation_succession_law_trigger = yes
	}
	can_pass = {
		can_change_acclamation_succession_law_trigger = yes
	}
	can_have = {
		can_keep_roamn_acclamation_succession_law_trigger = yes
	}
	should_start_with = {
		has_ep3_dlc_trigger = yes
		government_has_mechanic = administrative
		OR = {
			government_has_flag = government_is_subroman_administrative
			government_has_flag = government_is_roman_administrative
		}
		is_independent_ruler = yes
	}
	succession = {
		order_of_succession = appointment
		appointment_type = bp_roman_emperor
	}
	flag = advanced_succession_law
	flag = can_designate_heirs
	flag = appointment_type_succession
	pass_cost = {
		prestige = {
			value = change_succession_law_prestige_cost
			multiply = {
				value = 0.5 # We reduce the cost to ensure that going back to acclamation is easy
				desc = "[administrative_government|E]"
			}
		}
	}
	ai_will_do = {
		if = {
			limit = { has_realm_law = confederate_partition_succession_law }
			value = 1
		}
	}
	potential = {
		government_has_mechanic = administrative
		is_independent_ruler = yes
		OR = {
			government_has_flag = government_is_subroman_administrative
			government_has_flag = government_is_roman_administrative
		}
	}
}

roman_appointment_succession_law = {
	law_group_type = succession_order_laws
	index = 36

	can_keep = {
		can_keep_roman_appointment_succession_law_trigger = yes
		# Vassal Contract allows it
		custom_description = {
			text = succession_laws_must_not_be_forced_partition
			NAND = {
				top_liege != this
				vassal_contract_has_flag = locked_to_partition
			}
		}
	}
	can_have = {
		can_have_roman_appointment_succession_law_trigger = yes
	}
	can_pass = {
		can_change_appointment_succession_law_trigger = yes
	}
	should_start_with = {
		has_ep3_dlc_trigger = yes
		government_has_mechanic = administrative
		OR = {
			government_has_flag = government_is_subroman_administrative
			government_has_flag = government_is_roman_administrative
		}
		is_independent_ruler = no
	}
	succession = {
		order_of_succession = appointment
		appointment_type = bp_roman_governor
	}
	flag = advanced_succession_law
	flag = appointment_type_succession
	modifier = {
		glory_hound_opinion = -10
		parochial_opinion = -5
		player_heir_opinion = 20
		eligible_child_except_player_heir_opinion = -5
	}
	pass_cost = {
		prestige = change_succession_law_prestige_cost
	}
	revoke_cost = {
		prestige = change_succession_law_prestige_cost
	}

	ai_will_do = {
		if = {
			limit = {
				OR = {
					has_realm_law = confederate_partition_succession_law
					has_realm_law = partition_succession_law
					has_realm_law = high_partition_succession_law
				}
			}
			value = 3
		}
	}
	potential = {
		government_has_mechanic = administrative
		is_independent_ruler = no
		OR = {
			government_has_flag = government_is_subroman_administrative
			government_has_flag = government_is_roman_administrative
		}
	}
}
