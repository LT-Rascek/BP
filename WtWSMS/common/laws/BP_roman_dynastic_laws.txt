@roman_dynastic_authority_cooldown_years = 10

roman_dynastic_authority_0 = {
	law_group_type = roman_dynastic_authority
	index = 0

	modifier = {
		vassal_tax_contribution_mult = 0.05
		vassal_levy_contribution_mult = 0.05
	}
	flag = authority_level_0
	flag = title_revocation_allowed
	flag = titles_cannot_leave_realm_on_succession # Hardcoded flag
	flag = can_have_tributaries
	flag = enables_successor_roman_county_wars
	
	can_keep = {
		realm_law_use_roman_dynastic_authority_trigger = yes
		realm_law_match_liege_law_level_subroman_dynastic_trigger = { LAW = roman_dynastic_authority_0 }
	}
	
	can_pass = {
		realm_law_vassal_has_liege_law_level_subroman_dynastic_trigger = { LAW = roman_dynastic_authority_0 }
	}
	
	pass_cost = {}
	
	on_pass = {
		#Remove modifiers
		remove_law_related_modifiers_effect = yes
		set_variable = {
			name = crown_authority_cooldown
			years = @roman_dynastic_authority_cooldown_years
		}
	}
}
roman_dynastic_authority_1 = {
	law_group_type = roman_dynastic_authority
	index = 1

	modifier = {
		glory_hound_opinion = -10
		parochial_opinion = -10
		courtly_opinion = -10
		minority_opinion = -20
		vassal_tax_contribution_mult = 0.10
		vassal_levy_contribution_mult = 0.10
	}
	flag = authority_level_1
	flag = title_revocation_allowed
	flag = titles_cannot_leave_realm_on_succession # Hardcoded flag
	flag = can_have_tributaries
	flag = enables_imperial_roman_county_wars
	flag = enables_successor_roman_duchy_county_wars
	
	can_keep = {
		realm_law_use_roman_dynastic_authority_trigger = yes
		realm_law_match_liege_law_level_subroman_dynastic_trigger = { LAW = roman_dynastic_authority_1 }
	}
	
	can_pass = {
		trigger_if = {
			limit = { has_realm_law = roman_dynastic_authority_0 }
			custom_description = {
				subject = root
				text = "has_roman_dynastic_authority_cooldown"
				NOT = { has_variable = crown_authority_cooldown }
			}
		}
		realm_law_vassal_has_liege_law_level_subroman_dynastic_trigger = { LAW = roman_dynastic_authority_1 }
	}
	
	pass_cost = {
		prestige = {
			if = {
				limit = {
					NOR = {
						has_realm_law = roman_dynastic_authority_2
						has_realm_law = roman_dynastic_authority_3
					}
				}
				add = increase_crown_authority_prestige_cost
			}
		}
	}
	
	on_pass = {
		#Remove modifiers
		remove_law_related_modifiers_effect = yes
		set_variable = {
			name = crown_authority_cooldown
			years = @roman_dynastic_authority_cooldown_years
		}
	}
}
