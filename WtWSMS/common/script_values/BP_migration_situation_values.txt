cultural_unification_cb_prestige_cost = {
	value = 0
	if = {
		limit = {
			NAND = {
				scope:attacker.faith.religious_head = scope:attacker
				scope:defender.faith = scope:attacker.faith
			}
		}
		add = {
			value = 200
			desc = CB_BASE_COST
		}
	}
	if = {
		limit = {
			exists = scope:defender
			scope:attacker.primary_title.tier < scope:defender.primary_title.tier
		}
		multiply = {
			value = 1.5
			desc = CB_COST_ATTACKING_SUPERIOR
		}
	}
	if = {
		limit = {
			exists = scope:defender
			scope:defender ?= scope:attacker.culture.culture_head
		}
		multiply = {
			value = 1.5
			desc = CB_COST_ATTACKING_CULTURE_HEAD
		}
	}
	multiply = common_cb_prestige_cost_multiplier
}
