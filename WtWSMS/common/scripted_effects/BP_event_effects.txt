### Secular Decisions
form_frankish_kingdom_effect = {
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:k_franks = {
		change_title_holder = {
			holder = root
			change = scope:change
		}
	}
	#Shift the capital Duchy, always
	this.capital_county.duchy = { set_de_jure_liege_title = title:k_franks }
	#Shift every duchy that's completely controlled
	every_realm_county = { this.duchy = { add_to_list = possible_duchies } }
	every_in_list = { #If it's completely controlled and neighboring the capital, shift its de jure liege
		list = possible_duchies
		limit = {
			root = { completely_controls = prev.duchy }
			any_county = {
				this.duchy = prev
				any_neighboring_county = { this.duchy = root.capital_county.duchy }
			}
		}
		set_de_jure_liege_title = title:k_franks
	}
	resolve_title_and_vassal_change = scope:change
	
	#Make the kingdom of the Franks part of some de jure empire
	hidden_effect = {
		title:k_franks = { set_de_jure_liege_title = title:e_praetorian_prefecture_gaul } #Whooo-boy!
	}
	
	#Move capital to Aachen if its not held by a human player
	if = {
		limit = {
			any_realm_county = { this = title:c_aachen }
			title:c_aachen.holder = { is_ai = yes }
		}
		create_title_and_vassal_change = {
			type = returned
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:c_aachen = {
			change_title_holder = {
				holder = root
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		set_realm_capital = title:c_aachen
	}
	else_if = {
		limit = { title:c_aachen.holder = root }
		set_realm_capital = title:c_aachen
	}
	
	#Give the special Soissons subjugation CB
	custom_description = {
		text = unlocks_one_major_cb_against_roman_state
		add_character_flag = available_realm_conquest_cb_against_centralized_state
	}
	
	#Add claims to Gaul
	add_pressed_claim = title:e_praetorian_prefecture_gaul
	if = {
		limit = { culture = { has_cultural_parameter = male_only_inheritance } }
		every_heir = {
			limit = { is_female = no }
			add_pressed_claim = title:e_praetorian_prefecture_gaul
		}
	}
	else_if = {
		limit = { culture = { has_cultural_parameter = female_only_inheritance } }
		every_heir = {
			limit = { is_male = no }
			add_pressed_claim = title:e_praetorian_prefecture_gaul
		}
	}
	else = { every_heir = { add_pressed_claim = title:e_praetorian_prefecture_gaul } }
}

### Formation of the Aures Kingdom
form_k_aures_effect = {
	title:k_auresia = {
		set_coa = title:d_auresia
		copy_title_history = title:d_auresia
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:k_auresia = {
		change_title_holder = {
			holder = root
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
	title:k_auresia.holder = { destroy_title = title:d_auresia }
	#Make the kingdom of Auresia part of some de jure empire
	hidden_effect = {
		title:k_auresia = { set_de_jure_liege_title = title:e_north_africa }
	}
}

### Create human sacrifice
establish_human_sacrifice_effect = {
	if = {
		limit = { scope:human_sacrifice_reformer.faith = faith:gothic_pagan }
		scope:human_sacrifice_reformer.faith = {
			remove_doctrine = tenet_ritual_celebrations
			add_doctrine = tenet_human_sacrifice
		}
	}
	if = {
		limit = { scope:human_sacrifice_reformer.faith = faith:anglo_pagan }
		scope:human_sacrifice_reformer.faith = {
			remove_doctrine = tenet_ritual_celebrations
			add_doctrine = tenet_human_sacrifice
		}
	}
	if = {
		limit = { scope:human_sacrifice_reformer.faith = faith:przeworsk_pagan }
		scope:human_sacrifice_reformer.faith = {
			remove_doctrine = tenet_ritual_celebrations
			add_doctrine = tenet_human_sacrifice
		}
	}
	if = {
		limit = { scope:human_sacrifice_reformer.faith = faith:irimonic_pagan }
		scope:human_sacrifice_reformer.faith = {
			remove_doctrine = tenet_ritual_celebrations
			add_doctrine = tenet_human_sacrifice
		}
	}
	if = {
		limit = { scope:human_sacrifice_reformer.faith = faith:frankish_pagan }
		scope:human_sacrifice_reformer.faith = {
			remove_doctrine = tenet_ritual_celebrations
			add_doctrine = tenet_human_sacrifice
		}
	}
	hidden_effect = {
		scope:human_sacrifice_reformer.faith = {
			set_variable = { name = human_sacrifice_added years = 100 }
		}
	}
}

### Create Warmonger Tenet
establish_warmonger_effect = {
	if = {
		limit = { scope:warmonger_reformer.faith = faith:anglo_pagan }
		scope:warmonger_reformer.faith = {
			remove_doctrine = tenet_sanctity_of_nature #It's the most common Germanic Faith tenet, so ditch that instead
			add_doctrine = tenet_warmonger
		}
	}
	if = {
		limit = { scope:warmonger_reformer.faith = faith:przeworsk_pagan }
		scope:warmonger_reformer.faith = {
			remove_doctrine = tenet_sanctity_of_nature #It's the most common Germanic Faith tenet, so ditch that instead
			add_doctrine = tenet_warmonger
		}
	}
	if = {
		limit = { scope:warmonger_reformer.faith = faith:irimonic_pagan }
		scope:warmonger_reformer.faith = {
			remove_doctrine = tenet_unrelenting_faith
			add_doctrine = tenet_warmonger
		}
	}
}

### Create Bolghar culture
create_bolghar_culture_effect = {
	custom_description = {
		text = your_culture_adopts_the_bulghar_identity
		subject = root
		$THE_CULTURE$ = {
			set_culture_name = {
				noun = true_bolghar
				collective_noun = true_bolghar_collective_noun
				prefix = true_bolghar_prefix
			}
			set_name_list = name_list_bolghar
		}
	}
	
	#All your Oghuric Vassals adopt your culture, regardless of their own
	custom_description = {
		text = all_oghuric_vassals_adopt_your_culture
		subject = root
		
		every_vassal_or_below = {
			limit = {
				NOT = { this.culture = $THE_CULTURE$ }
				culture = { has_cultural_pillar = heritage_oghuric }
			}
			set_culture = $THE_CULTURE$
			every_courtier_or_guest = {
				limit = {
					NOT = { this.culture = $THE_CULTURE$ }
					culture = { has_cultural_pillar = heritage_oghuric }
				}
				set_culture = $THE_CULTURE$
			}
		}
		
		every_tributary = {
			limit = {
				NOT = { this.culture = $THE_CULTURE$ }
				culture = { has_cultural_pillar = heritage_oghuric }
			}
			set_culture = $THE_CULTURE$
			every_courtier_or_guest = {
				limit = {
					NOT = { this.culture = $THE_CULTURE$ }
					culture = { has_cultural_pillar = heritage_oghuric }
				}
				set_culture = $THE_CULTURE$
			}
			every_vassal_or_below = {
				limit = {
					NOT = { this.culture = $THE_CULTURE$ }
					culture = { has_cultural_pillar = heritage_oghuric }
				}
				set_culture = $THE_CULTURE$
			}
		}
	}
	
	custom_description = {
		text = all_oghuric_vassal_counties_adopt_your_culture
		subject = root
		
		every_realm_county = {
			limit = {
				NOT = { this.culture = $THE_CULTURE$ }
				culture = { has_cultural_pillar = heritage_oghuric }
			}
			set_county_culture = $THE_CULTURE$
		}
		
		every_tributary = {
			limit = {
				NOT = { this.culture = $THE_CULTURE$ }
				culture = { has_cultural_pillar = heritage_oghuric }
			}
			every_realm_county = {
				limit = {
					NOT = { this.culture = $THE_CULTURE$ }
					culture = { has_cultural_pillar = heritage_oghuric }
				}
				set_county_culture = $THE_CULTURE$
			}
		}
	}
	
	custom_description = {
		text = nearby_oghur_rulers_adopt_your_culture
		subject = root
		
		every_ruler = {
			limit = {
				is_landed = yes
				NOT = { this.culture = $THE_CULTURE$ }
				culture = { has_cultural_pillar = heritage_oghuric }
				NOT = { root = { target_is_vassal_or_below = prev } }
				NOT = { is_tributary_of = root }
				culture = { general_culture_will_not_flip_culture_trigger = no }
				#Distance Calculation; scope issues making script values difficult
				root = { realm_to_title_distance_squared = { target = prev.primary_title value <= root.prestige_distance_oghuric } }
			}
			set_culture = $THE_CULTURE$
			every_courtier_or_guest = {
				limit = {
					NOT = { this.culture = $THE_CULTURE$ }
					culture = { has_cultural_pillar = heritage_oghuric }
					culture = { general_culture_will_not_flip_culture_trigger = no }
				}
				set_culture = $THE_CULTURE$
			}
			every_vassal_or_below = {
				limit = {
					NOT = { this.culture = $THE_CULTURE$ }
					culture = { has_cultural_pillar = heritage_oghuric }
					culture = { general_culture_will_not_flip_culture_trigger = no }
				}
				set_culture = $THE_CULTURE$
			}
		}
	}
	
	#Flip Counties
	custom_description = {
		text = nearby_oghur_counties_adopt_your_culture
		subject = root
		
		every_county = {
			limit = {
				is_landless_type_title = no
				this.culture = { has_cultural_pillar = heritage_oghuric }
				NOT = { this.culture = $THE_CULTURE$ }
				NOT = { root = { target_is_vassal_or_below = prev.holder } }
				NOT = { this.holder = { is_tributary_of = root } }
				culture = { general_culture_will_not_flip_culture_trigger = no }
				#Distance Calculation; scope issues making script values difficult
				root = { realm_to_title_distance_squared = { target = prev.primary_title value <= root.prestige_distance_oghuric } }
			}
			set_county_culture = $THE_CULTURE$
		}
	}
	
	hidden_effect = { set_global_variable = bolghar_idenity_formed }
}
