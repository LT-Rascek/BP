tenet_redeeming_struggle = {
	icon = tenet_unrelenting_faith

	name = tenet_redeeming_struggle_name

	desc = tenet_redeeming_struggle_desc

	is_shown = {}

	can_pick = {
		religion = religion:christianity_religion
		custom_description = {
			text = doctrine_inverted_pluralism_fundamentalist_trigger
			doctrine:doctrine_pluralism_fundamentalist = { is_in_list = selected_doctrines }
		}
	}
	can_pick_as_personal_tenet = {
		NOR = {
			has_personal_tenet = tenet_pacifism
			has_personal_tenet = tenet_dharmic_pacifism
			has_personal_tenet = tenet_communal_possessions
			has_personal_tenet = tenet_unrelenting_faith
		}
	}

	piety_cost = {
		value = faith_tenet_cost_high

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_redeeming_struggle }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	divergence_multiplier = 1.25

	special_parameters = {
		minimum_fervor = 25
	}

	character_modifier = {
		monthly_county_control_growth_factor = -0.2
	}

	parameters = {
		tenet_redeeming_struggle_resistance = yes
		great_holy_wars_active = yes
		peasant_and_populist_factions_more_common = yes
	}

	personal_tenet_modifier = {
		monthly_county_control_growth_factor = -0.1
		tolerance_advantage_mod = 5
		county_opinion_add = 10
	}

	personal_tenet_parameters = {
		cheaper_holy_wars_active = yes
		raise_circumcellions_decision = yes
	}

	traits = {
		virtues = {
			peasant_leader = { weight = 15 scale = 2 }
		}
		sins = {
			arrogant = { weight = 10 scale = 1.5 }
		}
	}
}
