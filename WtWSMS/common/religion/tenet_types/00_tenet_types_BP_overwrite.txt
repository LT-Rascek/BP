tenet_consolamentum = {
	icon = tenet_consolamentum

	name = {
		first_valid = {
			##### Changed from Vanilla vvvvv
			triggered_desc = {
				trigger = {
					NOT = { rite_religion_tag = christianity_religion }
					rite_has_parameter = liturgical_language_greek
				}
				desc = tenet_consolamentum_nonchristian_greek_name
			}
			triggered_desc = {
				trigger = {
					NOT = { rite_religion_tag = christianity_religion }
					rite_has_parameter = liturgical_language_latin
				}
				desc = tenet_consolamentum_nonchristian_latin_name
			}
			##### Changed from Vanilla ^^^^^
			triggered_desc = {
				trigger = {
					has_pam_dlc_trigger = yes
					pam_latin_as_liturgical_language_trigger = yes
				}
				desc = tenet_consolamentum_latin_name
			}
			triggered_desc = {
				trigger = {
					has_pam_dlc_trigger = yes
					pam_greek_as_liturgical_language_trigger = yes
				}
				desc = tenet_consolamentum_greek_name
			}
			triggered_desc = {
				trigger = {
					has_pam_dlc_trigger = yes
					pam_greek_as_liturgical_language_trigger = no
					pam_latin_as_liturgical_language_trigger = no
				}
				desc = tenet_consolamentum_default_name
			}
			triggered_desc = {
				trigger = {
					has_pam_dlc_trigger = no
					rite_religion_tag = christianity_religion
				}
				desc = tenet_consolamentum_name
			}
			desc = tenet_consolamentum_name_alternate
		}
	}

	desc = {
		first_valid = {
			##### Changed from Vanilla vvvvv
			triggered_desc = {
				trigger = {
					NOT = { rite_religion_tag = christianity_religion }
					OR = {
						rite_has_parameter = liturgical_language_latin
						rite_has_parameter = liturgical_language_greek
					}
				}
				desc = tenet_consolamentum_nonchristian_latin_name
			}
			##### Changed from Vanilla ^^^^^
			triggered_desc = {
				trigger = {
					has_pam_dlc_trigger = yes
					pam_latin_as_liturgical_language_trigger = yes
				}
				desc = tenet_consolamentum_latin_desc
			}
			triggered_desc = {
				trigger = {
					has_pam_dlc_trigger = yes
					pam_greek_as_liturgical_language_trigger = yes
				}
				desc = tenet_consolamentum_greek_desc
			}
			triggered_desc = {
				trigger = {
					has_pam_dlc_trigger = yes
					pam_greek_as_liturgical_language_trigger = no
					pam_latin_as_liturgical_language_trigger = no
				}
				desc = tenet_consolamentum_default_desc
			}
			desc = tenet_consolamentum_desc
		}
	}

	piety_cost = {
		if = {
			limit = {
				rite_has_tenet = tenet_sacrificial_ceremonies
			}
			value = faith_tenet_cost_low
		}
		else_if = {
			limit = {
				OR = {
					rite_religion_tag = dualism_religion
					rite_religion_tag = hinduism_religion
					rite_religion_tag = jainism_religion
				}
			}
			value = faith_tenet_cost_high
		}
		else = {
			value = faith_tenet_cost_mid
		}

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_consolamentum }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	can_pick = {
		NOR = {
			religion = religion:islam_religion
			religion = religion:judaism_religion
			religion = religion:zoroastrianism_religion
			AND = {
				has_doctrine = pagan_hostility_doctrine
				NOT = { religion = religion:kushitism_religion }
			}
		}
		custom_description = {
			text = incompatible_tenet_sacrificial_ceremonies_trigger
			NOT = { tenet:tenet_sacrificial_ceremonies = { is_in_list = selected_doctrines } }
		}
	}

	parameters = {
		consolamentum_active
	}

	divergence_multiplier = 2

	character_modifier = {
		short_reign_duration_mult = -0.5
	}

	personal_tenet_parameters = {
		consolamentum_grief_immunity
	}

	traits = {
		virtues = {
			chaste = { weight = 15 scale = 1 }
		}
		sins = {
			lustful = { weight = 15 scale = 1 }
		}
	}
}
