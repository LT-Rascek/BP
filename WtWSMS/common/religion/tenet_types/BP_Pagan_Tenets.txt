tenet_hero_cults = {
	icon = tenet_two_nature

	name = tenet_dyophysitism_name

	desc = tenet_dyophysitism_desc

	is_shown = {}

	can_pick = {
		religion = religion:christianity_religion
		unique_christology_trigger = yes
	}
	can_pick_as_personal_tenet = {
		unique_christology_personal_trigger = yes
	}

	piety_cost = {
		value = faith_tenet_cost_low

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_hero_cults }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	divergence_multiplier = 1.5

	character_modifier = {
		different_faith_opinion = 15
	}

	parameters = {}

	personal_tenet_parameters = {}

	traits = {
		# Doctrinal fidelity to the one-united-nature formula vs. dissimulation under Chalcedonian pressure.
		virtues = {
			humble = { weight = 15 scale = 1 }
		}
		sins = {
			arbitrary = { weight = 15 scale = 1 }
		}
	}
}
