##### Changed from Vanilla: Tied into Messianic Doctrine
special_doctrine_is_christian_faith = {
	doctrine_group_type = is_christian_faith
	icon = "special_doctrine_teachings_of_jesus"

	parameters = {
		special_doctrine_is_christian_faith
		christian_syncretic_recipient
		has_clerical_regions
	}

	special_parameters = {
		# Since doctrines cannot modify other faith's views of us, only our view of others, all Christian faiths are given this special doctrine which makes them friendlier to Faiths with the Christian Syncretism tenet.
		hostility_level_override = {
			temple_authority_messianic_doctrine = hostile
 			tenet_christian_syncretism = astray
			# We don't localise these because Adoptionists are supposed to be a bit of a surprise.
			special_doctrine_adoptionist_school_felicitous_doctrine = evil
			special_doctrine_adoptionist_school_radical_doctrine = evil
		}
		parameter_opinion_override = {
			christian_syncretic_actor = 30
		}
	}
}
