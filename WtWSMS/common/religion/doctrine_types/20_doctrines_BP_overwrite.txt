doctrine_monotheist = {
	doctrine_group_type = doctrine_theism
	index = 0
	divergence = 0
	
	icon = "doctrine_monotheist"
	
	piety_cost = {
		value = faith_doctrine_cost_high
		if = {
			limit = {
				OR = {
					rite_has_doctrine = abrahamic_hostility_doctrine
					rite_has_doctrine = mazdan_hostility_doctrine
					rite_has_doctrine = doctrine_monotheist
				}
			}
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_henotheist }
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_polytheist }
			multiply = faith_changed_doctrine_cost_mult_three_step
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_monist }
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_agnostic }
			multiply = faith_changed_doctrine_cost_mult_three_step
		}
	}
	
	special_parameters = {
		hostility_levels = {
			unrelated = evil
		}
	}
}
