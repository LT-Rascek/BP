#Migration GUI triggers
holder_or_above_is_migration_target_trigger = {
	holder = { save_temporary_scope_as = the_holder }
	OR = {
		holder = {
			general_situation_regional_parameter_trigger = {
				CHARACTER = scope:the_holder
				PARAMETER = targetable_by_migration_cbs
			}
		}
		trigger_if = {
			limit = { scope:the_holder = { is_independent_ruler = no } }
			scope:the_holder.top_liege = {
				general_situation_regional_parameter_trigger = {
					CHARACTER = scope:the_holder.top_liege
					PARAMETER = targetable_by_migration_cbs
				}
			}
		}
	}
}
#Make sure this stays linked well to bp_migration_war_distace_tier_logic_trigger
bp_migration_war_distace_tier_logic_gui_trigger = {
	OR = {
		#100k is about the distance from Constantinople to Chios or nearly to the Danube
		# We base everything on this distance, keeping mind the actual distance is the square-root
		# Of this value (i.e., 10k = 100)
		trigger_if = {
			limit = { this.migration_war_effective_fame_level > 4 }
			scope:target ?= {
				any_in_de_jure_hierarchy = {
					tier = tier_county
					holder = { in_diplomatic_range = scope:the_character }
				}
				holder_or_above_is_migration_target_trigger = yes
			}
		}
		trigger_if = {
			limit = { this.migration_war_effective_fame_level > 3 }
			scope:target ?= {
				any_in_de_jure_hierarchy = {
					tier = tier_county
					squared_distance = { target = scope:the_character.capital_county value < 640000 } #800
				}
				holder_or_above_is_migration_target_trigger = yes
			}
		}
		trigger_if = {
			limit = { this.migration_war_effective_fame_level > 2 }
			scope:target ?= {
				any_in_de_jure_hierarchy = {
					tier = tier_county
					squared_distance = { target = scope:the_character.capital_county value < 160000 } #400
				}
				holder_or_above_is_migration_target_trigger = yes
			}
		}
		trigger_if = {
			limit = { this.migration_war_effective_fame_level > 1 }
			scope:target ?= {
				any_in_de_jure_hierarchy = {
					tier = tier_county
					squared_distance = { target = scope:the_character.capital_county value < 40000 } #200
				}
				holder_or_above_is_migration_target_trigger = yes
			}
		}
		trigger_if = {
			limit = { this.migration_war_effective_fame_level > 0 }
			scope:target ?= {
				any_in_de_jure_hierarchy = {
					tier = tier_county
					squared_distance = { target = scope:the_character.capital_county value < 10000 } #100
				}
				holder_or_above_is_migration_target_trigger = yes
			}
		}
		#Neighboring Kingdom is always valid
		scope:target ?= {
			any_in_de_jure_hierarchy = {
				tier = tier_county
				any_title_to_title_neighboring_and_across_water_county = {
					holder = { target_is_same_character_or_above = scope:the_character }
				}
			}
		}
	}
}
