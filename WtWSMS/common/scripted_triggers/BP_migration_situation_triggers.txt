#Culture/Religion Group Triggers
BP_can_be_migrator_cultural_religious_trigger = {
	OR = {
		faith.religion = religion:tengrism_religion
		faith.religion = religion:slavic_religion
		faith.religion = religion:germanic_religion
		culture = { has_cultural_parameter = heritage_group_slavic }
		culture = { has_cultural_parameter = heritage_kulturbund_scythian }
		culture = { has_cultural_parameter = heritage_group_germanic }
	}
}
#Government Group Triggers

	government_has_flag = government_can_migrate
}
BP_can_be_barbarian_polity_governmental_trigger = {
	government_has_flag = government_is_centralized
}
BP_can_be_civilized_polity_government_trigger = {
	government_has_flag = government_is_centralized
}
#CBs
general_bp_migration_constraints_trigger = {
	#Pass on targets who have less total development than what we currently have
	this.total_target_development <= scope:defender.total_realm_development
	#Pass on targets smaller than our current realm size
	this.target_size <= scope:attacker.realm_size
}
target_title_not_in_great_steppe_trigger = {
	NOT = {
		situation:the_great_steppe = {
			any_situation_county = {
				count >= 1
				this.$TIER$ = $TARGET_TITLE$
			}
		}
	}
}
