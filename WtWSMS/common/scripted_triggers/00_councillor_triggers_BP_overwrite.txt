can_be_chancellor_trigger = {
	can_be_councillor_basics_trigger = yes
	$COURT_OWNER$ ?= {
		NOT = { government_has_flag = government_is_nomadic }
	}
	NAND = {
		has_council_position = councillor_court_chaplain
		rite = { rite_has_parameter = clerical_appointment_fixed }
	}
	NOR = {
		has_council_position = councillor_spouse
		AND = {
			this = liege.diarch
			liege = { has_diarchy_active_parameter = diarchy_is_vizierate }
		}
	}
	can_be_councillor_gender_trigger = { COURT_OWNER = $COURT_OWNER$ }
	##### Changed from Vanilla vvvvv
	trigger_if = {
		limit = {
			liege ?= { government_has_flag = only_wuzurgan_can_be_councillors }
			liege ?= { can_have_wuzurgan_trigger = yes }
		}
		is_wuzurgan_trigger = yes
	}
	##### Changed from Vanilla ^^^^^
}

# expect scope:councillor_liege, scope:councillor
task_conversion_valid_county_trigger = {
	is_landless_type_title = no
	save_temporary_scope_as = county_check
	# Liege rite when the county will convert into the liege's faith
	trigger_if = {
		limit = {
			OR = {
				scope:councillor_liege = {
					tgp_has_access_to_ministry_trigger = yes
				}
				faith = scope:councillor_liege.faith
				scope:councillor.faith = scope:councillor_liege.faith
			}
		}
		custom_tooltip = {
			text = task_conversion.tt.county_rite_already_matches_liege
			rite != scope:councillor_liege.rite
		}
	}
	### Minority stuff ###
	trigger_else_if = {
		limit = {
			rite = scope:councillor_liege.rite
			scope:councillor_liege = { has_realm_law_flag = may_convert_faith }
		}
		custom_tooltip = {
			text = "not_has_expel_minorities_unlocked"
			county_has_minority_faith_trigger = yes
			scope:councillor_liege = { has_realm_law_flag = unlock_expel_religious_minorities }
		}
	}
	### End Minority stuff ###
	trigger_else = {
		custom_tooltip = {
			text = task_conversion.tt.county_rite_already_matches_chaplain
			rite != scope:councillor.rite
		}
	}

	trigger_if = {
		limit = {
			scope:councillor_liege = {
				is_alive = yes
				has_variable = hold_court_8120_religion_block
			}
		}
		custom_tooltip = {
			text = hold_court_8120_conversion_tt
			faith != scope:councillor_liege.var:hold_court_8120_religion_block
		}
	}
	custom_description = { # Vassal Contract forbids meddling by liege
		text = "is_protected_via_contract_self_or_liege"
		subject = holder
		# don't "protect" same faith in edge cases
		trigger_if = {
			limit = {
				scope:county_check.holder = { is_vassal_or_below_of = scope:councillor_liege }
				scope:councillor_liege.faith != scope:county_check.holder.faith
			}
			holder = {
				NOR = {
					vassal_contract_has_flag = religiously_protected
					any_liege_or_above = {
						is_vassal_or_below_of = scope:councillor_liege
						vassal_contract_has_flag = religiously_protected
					}
				}
			}
		}
	}
	trigger_if = {
		limit = {	
			var:cannot_be_converted_by_value ?= scope:councillor_liege		
		}
		custom_tooltip = {
			text = promised_to_not_convert_county	
			NOT = {
				var:cannot_be_converted_by_value ?= scope:councillor_liege	
			}
		}
	}
	trigger_if = {
		limit = {
			scope:councillor_liege = { is_ai = yes }
			OR = {
				faith = scope:councillor_liege.faith
				faith = scope:councillor.faith
			}
		}
		NOT = {
			rite = { has_variable = pam_rite_grace_period }
		}
	}
	trigger_if = {
		limit = { scope:councillor_liege = { is_ai = yes } }
		# Unreformed Pagans do not tend to even try conversion of non-unreformed faiths unless zealous
		trigger_if = {
			limit = {
				scope:councillor_liege = {
					faith = {
						has_doctrine_parameter = unreformed
					}
				}
			}
			OR = {
				scope:councillor_liege = {
					has_character_flag = vassal_directive_convert_faith
					vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }
				}
				scope:councillor_liege = { ai_zeal >= 50 }
				scope:county_check.faith = { has_doctrine_parameter = unreformed }
				scope:county_check = { # Always convert holy sites
					any_county_province = {
						barony = {
							is_holy_site_of = scope:councillor_liege.faith
						}
					}
				}
			}
		}
		# Do not convert Righteous faiths, unless disturbingly zealous
		trigger_if = {
			limit = {
				scope:councillor_liege = {
					faith = {
						faith_hostility_level = {
							target = scope:county_check.faith
							value <= faith_fully_accepted_level
						}
					}
				}
			}
			OR = {
				scope:councillor_liege = {
					has_character_flag = vassal_directive_convert_faith
					vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }
				}
				scope:councillor_liege = {
					ai_zeal >= 100
					ai_rationality <= -50
				}
				scope:county_check = { # Always convert holy sites
					any_county_province = {
						barony = {
							is_holy_site_of = scope:councillor_liege.faith
						}
					}
				}
			}
		}
		# Converting Astray faiths only happens during certain circumstances
		trigger_if = {
			limit = {
				scope:councillor_liege = {
					faith = {
						faith_hostility_level = {
							target = scope:county_check.faith
							value <= faith_astray_level
						}
					}
				}
			}
			trigger_if = {
				limit = {
					scope:councillor_liege = {
						has_tolerant_faith_or_culture_trigger = yes
					}
				}
				OR = {
					scope:councillor_liege = {
						has_character_flag = vassal_directive_convert_faith
						vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }
					}
					scope:councillor_liege = { ai_zeal >= 100 }
					scope:county_check = { county_opinion <= -25 }
					scope:county_check = { # Always convert holy sites
						any_county_province = {
							barony = {
								is_holy_site_of = scope:councillor_liege.faith
							}
						}
					}
				}
			}
			trigger_else_if = {
				limit = {
					scope:councillor_liege = {
						rite = { rite_has_doctrine = doctrine_pluralism_fundamentalist }
					}
				}
				OR = {
					scope:councillor_liege = {
						has_character_flag = vassal_directive_convert_faith
						vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }
					}
					scope:councillor_liege = { ai_zeal >= 0 }
					scope:county_check = { county_opinion <= -10 }
					scope:county_check = { # Always convert holy sites
						any_county_province = {
							barony = {
								is_holy_site_of = scope:councillor_liege.faith
							}
						}
					}
				}
			}
			trigger_else = {
				OR = {
					scope:councillor_liege = {
						has_character_flag = vassal_directive_convert_faith
						vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }
					}
					scope:councillor_liege = { ai_zeal >= 25 }
					scope:county_check = { county_opinion <= -15 }
					scope:county_check = { # Always convert holy sites
						any_county_province = {
							barony = {
								is_holy_site_of = scope:councillor_liege.faith
							}
						}
					}
				}
			}
		}
		# Ash'ari in Spain tend to not convert Muwhalladi often
		trigger_if = {
			limit = {
				scope:councillor_liege = {
					faith = faith:sunni
					capital_province ?= { geographical_region = world_europe_west_iberia }
					scope:county_check.rite = rite:maliki
				}
				scope:county_check = {
					title_province = { geographical_region = world_europe_west_iberia }
				}
			}
			OR = {
				scope:councillor_liege = {
					has_character_flag = vassal_directive_convert_faith
					vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }
				}
				scope:councillor_liege = { ai_zeal >= 50 }
				scope:county_check = { county_opinion <= -20 }
				scope:county_check = { # Always convert holy sites
					any_county_province = {
						barony = {
							is_holy_site_of = scope:councillor_liege.faith
						}
					}
				}
			}
		}
		# If a ruler has Jizya tax, they'll only convert the same *religious family* when ai_zeal is really high, they're converting unreformed pagans, counties in a kingdom with a Holy Site, or others of the same religion
		trigger_if = {
			limit = {
				scope:councillor_liege = {
					OR = {
						rite = { rite_has_tenet = tenet_tax_nonbelievers }
						faith = { has_doctrine = special_doctrine_jizya }
					}
				}
				scope:councillor_liege.faith = { faith_is_in_same_rf = { FAITH = scope:county_check.faith } }
			}
			OR = {
				scope:councillor_liege = {
					has_character_flag = vassal_directive_convert_faith
					vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }
				}
				scope:councillor_liege = { ai_zeal >= 50 }
				scope:county_check.faith = { has_doctrine_parameter = unreformed }
				scope:councillor_liege.faith = {
					any_holy_site_barony = {
						kingdom = scope:county_check.kingdom
					}
				}
				#scope:county_check.religion = scope:councillor_liege.religion # Excluded, as tax nonbelievers actually works for different faiths within the same religion
				AND = { # Islamic heartlands, to help fend off encroaching faiths
					scope:councillor_liege.religion = religion:islam_religion
					scope:county_check.title_province = {
						geographical_region = islamic_heartlands
					}
				}
			}
		}
	}
	# Struggle phases may sometimes restrict targets for involved characters.
	trigger_if = {
		# Give an exception for pre-existing conversions: these can take a while and it's not fair to lose years or decades of progress to a phase change.
		limit = {
			NOT = {
				scope:county_check = { is_target_of_council_task = task_conversion }
			}
		}
		NOT = {
			scope:councillor_liege = {
				custom_tooltip = {
					text = task_conversion.tt.forbidden.due_to_struggle_phase
					any_character_struggle = {
						involvement = involved
						is_struggle_parameter_active_secondary_county_trigger = {
							PARAMETER = county_conversion_disabled_for_involved_faiths_within_region
							COUNTY = scope:county_check
						}
					}
				}
			}
		}
	}
	##### Changed from Vanilla vvvvv
	custom_description = {
		text = "has_general_religious_protection_from_zoroastrian_cults"
		subject = holder
		object = scope:councillor_liege
		# The Shah has granted religious freedom to this faith, so holders are implicitly exempt
		NAND = {
			scope:councillor_liege.rite = { has_doctrine = doctrine_major_branch_dorustdin }
			holder ?= {
				is_ruler = yes
				is_independent_ruler = no
				rite = { has_doctrine = special_doctrine_legalized_faith_persia }
			}
		}
	}
	trigger_if = {
		limit = { NOT = { scope:councillor_liege.top_liege.culture = { has_innovation = innovation_state_religion } } }
		custom_description = {
			text = must_be_direct_vassal_without_state_religion
			subject = scope:councillor_liege
			object = holder
			holder ?= {
				OR = {
					is_vassal_of = scope:councillor_liege
					this = scope:councillor_liege
				}
			}
		}
		custom_description = {
			text = must_be_same_faith_or_tribal_if_top_liege_without_state_religion
			subject = scope:councillor_liege
			object = holder
			OR = {
				holder.rite = scope:councillor_liege.rite
				this.holder ?= {
					is_ruler = yes
					is_independent_ruler = no
					government_has_flag = government_is_tribal
				}
			}
		}
	}
	trigger_if = {
		limit = {
			scope:councillor_liege.rite = { has_doctrine = doctrine_major_branch_behdin }
			exists = scope:councillor_liege.rite.religious_head.rite
		}
		NOT = { this.rite.religious_head = scope:councillor_liege.rite.religious_head }
	}
	#Roman government minority rulers can't convert if they've aligned themselves with the Romans
	trigger_if = {
		limit = { scope:councillor_liege = { has_realm_law_flag = imperial_church_protected } }
		
		custom_description = {
			text = imperial_church_protected_state
			bp_county_has_roman_faith_trigger = no
		}
	}
