	$ACTOR$.culture = $TARGET$.faith.var:$VARIABLE$
}
actor_is_faith_child_culture_for_cultural_religions_trigger = {
	AND = {
		NOT = { $ACTOR$.culture = $TARGET$.faith.var:$VARIABLE$ }
		AND = {
			$ACTOR$.culture = { any_parent_culture_or_above = { this = $TARGET$.faith.var:$VARIABLE$ } }
			$ACTOR$.culture = { has_same_culture_heritage = $TARGET$.faith.var:$VARIABLE$ }
		}
	}
}
actor_is_same_heritage_for_cultural_religions_trigger = {
	AND = {
		NOT = { $ACTOR$.culture = $TARGET$.faith.var:$VARIABLE$ }
		NAND = {
			$ACTOR$.culture = { any_parent_culture_or_above = { this = $TARGET$.faith.var:$VARIABLE$ } }
			$ACTOR$.culture = { has_same_culture_heritage = $TARGET$.faith.var:$VARIABLE$ }
		}
		$ACTOR$.culture = { has_same_culture_heritage = $TARGET$.faith.var:$VARIABLE$ }
	}
}
actor_is_different_heritage_for_cultural_religions_trigger = {
	AND = {
		NOT = { $ACTOR$.culture = $TARGET$.faith.var:$VARIABLE$ }
		NAND = {
			$ACTOR$.culture = { any_parent_culture_or_above = { this = $TARGET$.faith.var:$VARIABLE$ } }
			$ACTOR$.culture = { has_same_culture_heritage = $TARGET$.faith.var:$VARIABLE$ }
		}
		NOT = { $ACTOR$.culture = { has_same_culture_heritage = $TARGET$.faith.var:$VARIABLE$ } }
	}
}
actor_is_faith_culture_or_child_culture_for_cultural_religions_trigger = {
	exists = $ACTOR$.faith.var:$VARIABLE$
	OR = {
		$ACTOR$.faith.var:$VARIABLE$ = $TARGET$.culture
		AND = {
			$ACTOR$.faith.var:$VARIABLE$ = { any_parent_culture_or_above = { this = $TARGET$.culture } }
			$ACTOR$.faith.var:$VARIABLE$ = { has_same_culture_heritage = $TARGET$.culture }
		}
	}
}

unique_soteriology_trigger = {
	custom_tooltip = {
		text = INCOMPATIBLE_TENET_OTHER_SOTERIOLOGY_TRIGGER
		NOR = {
			tenet:tenet_pelagianism = { is_in_list = selected_doctrines }
			tenet:tenet_total_depravity = { is_in_list = selected_doctrines }
			tenet:tenet_original_sin = { is_in_list = selected_doctrines }
			tenet:tenet_antinomianism = { is_in_list = selected_doctrines }
			tenet:tenet_gnosticism = { is_in_list = selected_doctrines }
		}
	}
}
#Changed from Vanilla: Extended to all WtWSMS Christologies
unique_soteriology_personal_trigger = {
	trigger_if = {
		limit = { has_global_variable = game_has_started }
		custom_tooltip = {
			text = INCOMPATIBLE_TENET_OTHER_SOTERIOLOGY_TRIGGER
			NOR = {
				has_personal_tenet = tenet_pelagianism
				has_personal_tenet = tenet_total_depravity
				has_personal_tenet = tenet_original_sin
				has_personal_tenet = tenet_antinomianism
				has_personal_tenet = tenet_gnosticism
			}
		}
	}
	trigger_else = {
		# prevent startup assert error
		custom_tooltip = {
			text = INCOMPATIBLE_TENET_OTHER_SOTERIOLOGY_TRIGGER
			calc_true_if = {
				amount <= 1
				has_personal_tenet = tenet_pelagianism
				has_personal_tenet = tenet_total_depravity
				has_personal_tenet = tenet_original_sin
				has_personal_tenet = tenet_antinomianism
				has_personal_tenet = tenet_gnosticism
			}
		}
	}
}

unique_eucharsitic_tenet_personal_trigger = {
	custom_tooltip = {
		text = INCOMPATIBLE_TENET_OTHER_COMMUNION_TRIGGER
		NOR = {
			tenet:tenet_memorialism = { is_in_list = selected_doctrines }
			tenet:tenet_repudiation = { is_in_list = selected_doctrines }
			#Handle having both Divine Office and Communion
			AND = {
				tenet:tenet_divine_liturgy = { is_in_list = selected_doctrines }
				tenet:tenet_communion = { is_in_list = selected_doctrines }
			}
		}
	}
}
